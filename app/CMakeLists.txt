cmake_minimum_required(VERSION 3.24)
project(CyGameSDK VERSION 0.29.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This foundation currently ships a Windows DX11 host.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Exact dependency identity, shared with the setup/build entry points.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock.json" CYGAME_DEPENDENCIES)
string(JSON CYGAME_CYUI_REVISION GET "${CYGAME_DEPENDENCIES}" CyUI revision)
string(JSON CYGAME_CYUI_REPOSITORY GET "${CYGAME_DEPENDENCIES}" CyUI repository)
set(CYUI_ROOT "" CACHE PATH "Optional clean checkout at the locked CyUI revision")
find_package(Git REQUIRED)
if(NOT CYUI_ROOT)
    include(FetchContent)
    set(FETCHCONTENT_BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/dependencies" CACHE PATH "Private dependency cache")
    FetchContent_Declare(cygame_cyui GIT_REPOSITORY "${CYGAME_CYUI_REPOSITORY}"
        GIT_TAG "${CYGAME_CYUI_REVISION}" GIT_SHALLOW FALSE GIT_SUBMODULES ""
        # Populate first; add the library only after host targets/modules are set.
        SOURCE_SUBDIR cygame_checkout_only)
    FetchContent_GetProperties(cygame_cyui)
    if(NOT cygame_cyui_POPULATED)
        FetchContent_MakeAvailable(cygame_cyui)
    endif()
    set(CYUI_ROOT "${cygame_cyui_SOURCE_DIR}")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${CYUI_ROOT}" -C "${CYUI_ROOT}" rev-parse HEAD
    OUTPUT_VARIABLE CYGAME_CYUI_ACTUAL OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT CYGAME_CYUI_ACTUAL STREQUAL CYGAME_CYUI_REVISION)
    message(FATAL_ERROR "CYUI_ROOT must be at revision ${CYGAME_CYUI_REVISION}; got ${CYGAME_CYUI_ACTUAL}")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${CYUI_ROOT}" -C "${CYUI_ROOT}" status --porcelain
    OUTPUT_VARIABLE CYGAME_CYUI_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(CYGAME_CYUI_DIRTY)
    message(FATAL_ERROR "Use a clean CyUI checkout; application migration must not modify the shared library")
endif()
set(IMGUI "${CYUI_ROOT}/third_party/imgui")
add_library(editor_imgui STATIC
    "${IMGUI}/imgui.cpp" "${IMGUI}/imgui_draw.cpp" "${IMGUI}/imgui_tables.cpp"
    "${IMGUI}/imgui_widgets.cpp"
    "${IMGUI}/backends/imgui_impl_win32.cpp" "${IMGUI}/backends/imgui_impl_dx11.cpp")
target_include_directories(editor_imgui PUBLIC "${IMGUI}" "${IMGUI}/backends")
target_link_libraries(editor_imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
set(CYUI_IMGUI_TARGET editor_imgui CACHE STRING "" FORCE)
set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_GALLERY OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_WEB_CEF OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
string(JSON CYGAME_MODULE_COUNT LENGTH "${CYGAME_DEPENDENCIES}" CyUI modules)
math(EXPR CYGAME_MODULE_LAST "${CYGAME_MODULE_COUNT} - 1")
set(CYGAME_MODULES "")
foreach(index RANGE ${CYGAME_MODULE_LAST})
    string(JSON module GET "${CYGAME_DEPENDENCIES}" CyUI modules ${index})
    list(APPEND CYGAME_MODULES "${module}")
endforeach()
set(CYUI_MODULES "${CYGAME_MODULES}" CACHE STRING "" FORCE)
add_library(editor_imguizmo STATIC third_party/imguizmo/ImGuizmo.cpp)
target_include_directories(editor_imguizmo PUBLIC third_party/imguizmo)
target_link_libraries(editor_imguizmo PUBLIC editor_imgui)
set(CYUI_IMGUIZMO_TARGET editor_imguizmo CACHE STRING "" FORCE)
add_subdirectory("${CYUI_ROOT}" "${CMAKE_BINARY_DIR}/cyui")
add_library(editor_core STATIC native/Core.cpp native/Scene3D.cpp native/ModelCommands.cpp native/ModelAuthoring.cpp native/JsonDocuments.cpp native/Build.cpp native/Assets.cpp native/Imports.cpp native/AssetReferences.cpp native/Scene.cpp native/Graph.cpp native/Blueprint.cpp native/Interfaces.cpp native/SignatureMigration.cpp native/TypeLibrary.cpp native/TypeMigration.cpp native/GraphCollapse.cpp native/GraphExpand.cpp native/Macros.cpp native/GraphRuntime.cpp native/Functions.cpp native/Materials.cpp native/MaterialProgram.cpp native/Image.cpp native/GraphBatch.cpp native/GraphNavigation.cpp native/GraphDiff.cpp)
target_include_directories(editor_core PUBLIC native third_party/nlohmann)
target_link_libraries(editor_core PUBLIC ole32 bcrypt windowscodecs)
add_executable(CyGameEditor WIN32 native/main.cpp native/MaterialPreview.cpp native/Editor.cpp native/MeshPreview.cpp native/ModelView.cpp native/ModelAuthoringView.cpp native/MotionToolsView.cpp native/JsonView.cpp native/InterfaceOverview.cpp native/BuildClient.cpp native/BuildView.cpp native/Content.cpp native/Outliner.cpp native/Workspace.cpp native/Viewport.cpp native/Gizmo.cpp native/GraphView.cpp native/GraphPanels.cpp native/GraphNavigatorView.cpp native/GraphDiffView.cpp native/FunctionsView.cpp native/BlueprintView.cpp native/TypesView.cpp native/InterfacesView.cpp native/SignatureMigrationView.cpp native/GraphCollapseView.cpp native/GraphExpandView.cpp native/GameCodecView.cpp native/MaterialInstanceView.cpp native/MaterialTools.cpp native/TextureView.cpp native/ImportView.cpp native/ThumbnailPreview.cpp native/TexturePreview.cpp native/ReferenceView.cpp native/Ipc.cpp)
target_include_directories(CyGameEditor PRIVATE third_party/imguizmo)
target_link_libraries(CyGameEditor PRIVATE editor_core CyUI::Core CyUI::Nodes CyUI::Inspector CyUI::Assets CyUI::Viewport CyUI::ImGuizmo CyUI::Timeline CyUI::Code CyUI::Commands CyUI::Diagnostics CyUI::Diff comdlg32 advapi32 shell32 user32 gdi32)
set(CYGAME_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Native editor output directory")
set_target_properties(CyGameEditor PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CYGAME_OUTPUT_DIR}$<0:>")
if(MSVC)
    foreach(target editor_core CyGameEditor editor_imgui editor_imguizmo)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc /MP)
        target_compile_definitions(${target} PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    endforeach()
endif()
enable_testing()
add_executable(CyGameSDKTests tests/core_tests.cpp native/Ipc.cpp)
target_link_libraries(CyGameSDKTests PRIVATE editor_core advapi32)
if(MSVC)
    target_compile_options(CyGameSDKTests PRIVATE /W4 /utf-8 /EHsc)
endif()
set_target_properties(CyGameSDKTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/build$<0:>")
add_test(NAME editor_contracts COMMAND CyGameSDKTests "${CMAKE_CURRENT_SOURCE_DIR}/..")
