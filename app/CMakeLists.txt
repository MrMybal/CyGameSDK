cmake_minimum_required(VERSION 3.24)
project(CyGameSDK VERSION 0.32.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This foundation currently ships a Windows DX11 host.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Exact dependency identity, shared with the setup/build entry points.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock.json" CYGAME_DEPENDENCIES)
string(JSON CYGAME_CYUI_REVISION GET "${CYGAME_DEPENDENCIES}" CyUI revision)
string(JSON CYGAME_CYUI_REPOSITORY GET "${CYGAME_DEPENDENCIES}" CyUI repository)
set(CYUI_ROOT "" CACHE PATH "Optional clean checkout at the locked CyUI revision")
find_package(Git REQUIRED)
if(NOT CYUI_ROOT)
    include(FetchContent)
    set(FETCHCONTENT_BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/dependencies" CACHE PATH "Private dependency cache")
    FetchContent_Declare(cygame_cyui GIT_REPOSITORY "${CYGAME_CYUI_REPOSITORY}"
        GIT_TAG "${CYGAME_CYUI_REVISION}" GIT_SHALLOW FALSE GIT_SUBMODULES ""
        # Populate first; add the library only after host targets/modules are set.
        SOURCE_SUBDIR cygame_checkout_only)
    FetchContent_GetProperties(cygame_cyui)
    if(NOT cygame_cyui_POPULATED)
        FetchContent_MakeAvailable(cygame_cyui)
    endif()
    set(CYUI_ROOT "${cygame_cyui_SOURCE_DIR}")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${CYUI_ROOT}" -C "${CYUI_ROOT}" rev-parse HEAD
    OUTPUT_VARIABLE CYGAME_CYUI_ACTUAL OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT CYGAME_CYUI_ACTUAL STREQUAL CYGAME_CYUI_REVISION)
    message(FATAL_ERROR "CYUI_ROOT must be at revision ${CYGAME_CYUI_REVISION}; got ${CYGAME_CYUI_ACTUAL}")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${CYUI_ROOT}" -C "${CYUI_ROOT}" status --porcelain
    OUTPUT_VARIABLE CYGAME_CYUI_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(CYGAME_CYUI_DIRTY)
    message(FATAL_ERROR "Use a clean CyUI checkout; application migration must not modify the shared library")
endif()
set(IMGUI "${CYUI_ROOT}/third_party/imgui")
add_library(editor_imgui STATIC
    "${IMGUI}/imgui.cpp" "${IMGUI}/imgui_draw.cpp" "${IMGUI}/imgui_tables.cpp"
    "${IMGUI}/imgui_widgets.cpp"
    "${IMGUI}/backends/imgui_impl_win32.cpp" "${IMGUI}/backends/imgui_impl_dx11.cpp")
target_include_directories(editor_imgui PUBLIC "${IMGUI}" "${IMGUI}/backends")
target_link_libraries(editor_imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
set(CYUI_IMGUI_TARGET editor_imgui CACHE STRING "" FORCE)
set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_MARKDOWN_DEMO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_PORTABLE_DEMO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_GALLERY OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_WEB_CEF OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
string(JSON CYGAME_MODULE_COUNT LENGTH "${CYGAME_DEPENDENCIES}" CyUI modules)
math(EXPR CYGAME_MODULE_LAST "${CYGAME_MODULE_COUNT} - 1")
set(CYGAME_MODULES "")
foreach(index RANGE ${CYGAME_MODULE_LAST})
    string(JSON module GET "${CYGAME_DEPENDENCIES}" CyUI modules ${index})
    list(APPEND CYGAME_MODULES "${module}")
endforeach()
set(CYUI_MODULES "${CYGAME_MODULES}" CACHE STRING "" FORCE)
add_library(editor_imguizmo STATIC third_party/imguizmo/ImGuizmo.cpp)
target_include_directories(editor_imguizmo PUBLIC third_party/imguizmo)
target_link_libraries(editor_imguizmo PUBLIC editor_imgui)
set(CYUI_IMGUIZMO_TARGET editor_imguizmo CACHE STRING "" FORCE)
add_subdirectory("${CYUI_ROOT}" "${CMAKE_BINARY_DIR}/cyui")
# Native automation SDK; prefer a pinned local source checkout.
string(JSON CYGAME_CYMCP_REVISION GET "${CYGAME_DEPENDENCIES}" CyMCPSDK revision)
set(CYMCP_ROOT "" CACHE PATH "Clean CyMCPSDK source checkout at the locked revision")
if(NOT CYMCP_ROOT AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../CyMCPSDK/CMakeLists.txt")
    get_filename_component(CYMCP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../CyMCPSDK" ABSOLUTE)
endif()
if(NOT CYMCP_ROOT)
    include(FetchContent)
    FetchContent_Declare(cygame_cymcp GIT_REPOSITORY https://github.com/MrMybal/CyMCPSDK.git
        GIT_TAG "${CYGAME_CYMCP_REVISION}" GIT_SHALLOW FALSE GIT_SUBMODULES "" SOURCE_SUBDIR cygame_checkout_only)
    FetchContent_MakeAvailable(cygame_cymcp)
    set(CYMCP_ROOT "${cygame_cymcp_SOURCE_DIR}")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${CYMCP_ROOT}" -C "${CYMCP_ROOT}" rev-parse HEAD
    OUTPUT_VARIABLE CYGAME_CYMCP_ACTUAL OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT CYGAME_CYMCP_ACTUAL STREQUAL CYGAME_CYMCP_REVISION)
    message(FATAL_ERROR "CyMCPSDK must match ${CYGAME_CYMCP_REVISION}; select the pinned checkout with CYMCP_ROOT")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${CYMCP_ROOT}" -C "${CYMCP_ROOT}" status --porcelain
    OUTPUT_VARIABLE CYGAME_CYMCP_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(CYGAME_CYMCP_DIRTY)
    message(FATAL_ERROR "Use a clean CyMCPSDK checkout; do not modify the sibling SDK")
endif()
foreach(module HTTP PLATFORM MCP AI)
    set(CYMCP_BUILD_${module} ON CACHE BOOL "" FORCE)
endforeach()
foreach(module LOCAL EXAMPLES TESTS CHAT_APP BENCHMARKS)
    set(CYMCP_BUILD_${module} OFF CACHE BOOL "" FORCE)
endforeach()
set(CYMCP_BUILD_CHAT_UI ON CACHE BOOL "" FORCE)
add_subdirectory("${CYMCP_ROOT}" "${CMAKE_BINARY_DIR}/cymcp")
add_library(editor_automation_runtime STATIC native/automation/Runtime.cpp)
target_link_libraries(editor_automation_runtime PRIVATE CyMCP::MCP crypt32)
target_include_directories(editor_automation_runtime PRIVATE native/automation)
add_library(editor_assistant_ui STATIC native/automation/Assistant.cpp)
target_link_libraries(editor_assistant_ui PRIVATE editor_automation_runtime CyMCP::ChatUI CyMCP::MCP)
if(MSVC)
    target_compile_options(editor_assistant_ui PRIVATE /W4 /utf-8 /EHsc)
endif()
add_executable(CyGameCLI native/automation/Cli.cpp)
target_link_libraries(CyGameCLI PRIVATE editor_automation_runtime CyMCP::MCP)
if(MSVC)
    target_compile_options(editor_automation_runtime PRIVATE /W4 /utf-8 /EHsc)
    target_compile_options(CyGameCLI PRIVATE /W4 /utf-8 /EHsc)
endif()
add_library(editor_source_ui STATIC native/SourceEditorBackend.cpp)
target_link_libraries(editor_source_ui PRIVATE CyUI::CodeEditor)
target_link_libraries(editor_assistant_ui PRIVATE editor_source_ui)
target_include_directories(editor_source_ui PRIVATE native)
add_library(editor_core STATIC native/Core.cpp native/Scene3D.cpp native/ModelCommands.cpp native/ModelAuthoring.cpp native/JsonDocuments.cpp native/Build.cpp native/Assets.cpp native/Imports.cpp native/AssetReferences.cpp native/Scene.cpp native/Graph.cpp native/Blueprint.cpp native/Interfaces.cpp native/SignatureMigration.cpp native/TypeLibrary.cpp native/TypeMigration.cpp native/GraphCollapse.cpp native/GraphExpand.cpp native/Macros.cpp native/GraphRuntime.cpp native/Functions.cpp native/Materials.cpp native/MaterialProgram.cpp native/Image.cpp native/GraphBatch.cpp native/GraphNavigation.cpp native/GraphDiff.cpp)
target_include_directories(editor_core PUBLIC native third_party/nlohmann)
target_link_libraries(editor_core PUBLIC ole32 bcrypt windowscodecs)
add_executable(CyGameEditor WIN32 native/main.cpp native/Documentation.cpp native/MaterialPreview.cpp native/Editor.cpp native/AssistantView.cpp native/UiPropertyAdapter.cpp native/SourceEditor.cpp native/EditorCommands.cpp native/MeshPreview.cpp native/ModelView.cpp native/ModelAuthoringView.cpp native/MotionToolsView.cpp native/JsonView.cpp native/InterfaceOverview.cpp native/BuildClient.cpp native/BuildView.cpp native/Content.cpp native/Outliner.cpp native/Workspace.cpp native/Viewport.cpp native/Gizmo.cpp native/GraphView.cpp native/GraphPanels.cpp native/GraphNavigatorView.cpp native/GraphDiffView.cpp native/FunctionsView.cpp native/BlueprintView.cpp native/TypesView.cpp native/InterfacesView.cpp native/SignatureMigrationView.cpp native/GraphCollapseView.cpp native/GraphExpandView.cpp native/GameCodecView.cpp native/MaterialInstanceView.cpp native/MaterialTools.cpp native/TextureView.cpp native/ImportView.cpp native/ThumbnailPreview.cpp native/TexturePreview.cpp native/ReferenceView.cpp native/Ipc.cpp)
target_include_directories(CyGameEditor PRIVATE third_party/imguizmo)
target_link_libraries(CyGameEditor PRIVATE editor_core CyUI::Core CyUI::Nodes CyUI::Inspector CyUI::Assets CyUI::Viewport CyUI::ImGuizmo CyUI::Timeline CyUI::Code CyUI::Commands CyUI::Diagnostics CyUI::Diff editor_source_ui CyUI::Hierarchy CyUI::DataView CyUI::FilePicker editor_assistant_ui advapi32 shell32 user32 gdi32)
set(CYGAME_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Native editor output directory")
set_target_properties(CyGameEditor CyGameCLI PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CYGAME_OUTPUT_DIR}$<0:>")
find_program(CYGAME_POWERSHELL NAMES pwsh powershell REQUIRED)
option(CYGAME_BUILD_DOCUMENTATION "Regenerate CyDoc (OFF reuses the complete existing export for UI development)" ON)
if(CYGAME_BUILD_DOCUMENTATION)
add_custom_target(CyGameDocumentation ALL
    COMMAND "${CYGAME_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/build_documentation.ps1"
        -OutputDirectory "${CYGAME_OUTPUT_DIR}/Documentation"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.."
    COMMENT "Generate the complete local CyDoc reader" VERBATIM)
else()
    add_custom_target(CyGameDocumentation ALL
        COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/../Documentation" "${CYGAME_OUTPUT_DIR}/Documentation"
        COMMENT "Copy the existing complete CyDoc export" VERBATIM)
endif()
add_dependencies(CyGameEditor CyGameDocumentation)
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../Documentation/" DESTINATION Documentation COMPONENT Documentation)
if(MSVC)
    foreach(target editor_core CyGameEditor editor_imgui editor_imguizmo)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc /MP)
        target_compile_definitions(${target} PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    endforeach()
endif()
enable_testing()
add_executable(CyGameSDKTests tests/core_tests.cpp native/Ipc.cpp)
target_link_libraries(CyGameSDKTests PRIVATE editor_core advapi32)
if(MSVC)
    target_compile_options(CyGameSDKTests PRIVATE /W4 /utf-8 /EHsc)
endif()
set_target_properties(CyGameSDKTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/build$<0:>")
add_test(NAME editor_contracts COMMAND CyGameSDKTests "${CMAKE_CURRENT_SOURCE_DIR}/..")


add_executable(CyGameUIContractsTests tests/ui_adapters_tests.cpp native/UiPropertyAdapter.cpp)
target_link_libraries(CyGameUIContractsTests PRIVATE editor_core CyUI::Inspector CyUI::DataView)
set_target_properties(CyGameUIContractsTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CYGAME_OUTPUT_DIR}$<0:>")
if(MSVC)
    target_compile_options(CyGameUIContractsTests PRIVATE /W4 /utf-8 /EHsc)
endif()
add_test(NAME ui_adapter_contracts COMMAND CyGameUIContractsTests)

add_executable(CyGameMCPProbe tests/mcp_probe.cpp)
target_link_libraries(CyGameMCPProbe PRIVATE CyMCP::MCP)
set_target_properties(CyGameMCPProbe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CYGAME_OUTPUT_DIR}$<0:>")
if(MSVC)
    target_compile_options(CyGameMCPProbe PRIVATE /utf-8 /EHsc)
endif()
